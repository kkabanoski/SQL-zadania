SELECT s.marka, s.model, MAX(sr.data_serwisu) AS data_ostatniego_przegladu FROM Samochody s LEFT JOIN Serwisy sr ON s.id_samochodu = sr.id_samochodu WHERE s.status = 'dostępny' GROUP BY s.id_samochodu, s.marka, s.model ORDER BY data_ostatniego_przegladu DESC;
SELECT k.imie, k.nazwisko, w.data_wypozyczenia, w.data_zwrotu,DATEDIFF(w.data_zwrotu, w.data_wypozyczenia) AS liczba_dni_wypozyczenia,w.kwota AS wartosc_wypozyczenia FROM Wypozyczenia w JOIN Klienci k ON w.id_klienta = k.id_klienta WHERE w.data_zwrotu IS NOT NULL AND w.data_wypozyczenia BETWEEN '2023-01-01' AND '2023-12-31' ORDER BY w.data_wypozyczenia ASC;
SELECT k.id_klienta,k.imie,k.nazwisko,COUNT(w.id_wypozyczenia) AS liczba_wypozyczen FROM Klienci k JOIN Wypozyczenia w ON k.id_klienta = w.id_klienta JOIN Samochody s ON w.id_samochodu = s.id_samochodu WHERE w.data_wypozyczenia >= DATE_SUB(CURDATE(), INTERVAL 1 YEAR) AND s.id_samochodu NOT IN ( SELECT id_samochodu FROM Serwisy GROUP BY id_samochodu HAVING SUM(koszt) > 5000 ) GROUP BY k.id_klienta, k.imie, k.nazwisko ORDER BY liczba_wypozyczen DESC;
SELECT Samochody.marka, Samochody.model, AVG(DATEDIFF(Wypozyczenia.data_zwrotu,Wypozyczenia.data_wypozyczenia)) AS srednia_liczba_dni,(SELECT SUM(koszt) FROM Serwisy WHERE Serwisy.id_samochodu = Samochody.id_samochodu) AS koszt_napraw FROM Samochody JOIN Wypozyczenia ON Samochody.id_samochodu = Wypozyczenia.id_samochodu GROUP BY Samochody.id_samochodu,Samochody.marka,Samochody.model HAVING COUNT(Wypozyczenia.id_wypozyczenia) > 5;

SELECT  Samochody.id_samochodu,Samochody.marka,Samochody.model,AVG(Opinie.ocena) AS srednia_ocena FROM Samochody JOIN Opinie ON Samochody.id_samochodu = Opinie.id_samochodu GROUP BY Samochody.id_samochodu, Samochody.marka, Samochody.model HAVING AVG(Opinie.ocena) < 4 ORDER BY srednia_ocena ASC;
SELECT Klienci.id_klienta,Klienci.imie,Klienci.nazwisko,COUNT(Wypozyczenia.id_wypozyczenia) AS liczba_wypozyczen FROM Klienci JOIN Wypozyczenia ON Klienci.id_klienta = Wypozyczenia.id_klienta WHERE Klienci.id_klienta NOT IN (SELECT id_klienta FROM Opinie WHERE id_klienta IS NOT NULL) GROUP BY Klienci.id_klienta, Klienci.imie, Klienci.nazwisko HAVING COUNT(Wypozyczenia.id_wypozyczenia) >= 3;
SELECT Samochody.id_samochodu,Samochody.marka,Samochody.model,Klienci.imie,Klienci.nazwisko,SUM(DATEDIFF(IFNULL(Wypozyczenia.data_zwrotu, CURDATE()),Wypozyczenia.data_wypozyczenia)) AS liczba_dni FROM Samochody JOIN Wypozyczenia ON Samochody.id_samochodu = Wypozyczenia.id_samochodu JOIN Klienci ON Wypozyczenia.id_klienta = Klienci.id_klienta WHERE YEAR(Klienci.data_rejestracji) BETWEEN 2022 AND 2023 AND Samochody.id_samochodu IN (SELECT id_samochodu FROM Wypozyczenia WHERE data_wypozyczenia >= DATE_SUB(CURDATE(), INTERVAL 12 MONTH)) GROUP BY Samochody.id_samochodu, Samochody.marka, Samochody.model, Klienci.id_klienta, Klienci.imie, Klienci.nazwisko ORDER BY liczba_dni DESC;
SELECT Klienci.imie,Klienci.nazwisko,Klienci.email,Faktury.id_faktury,Faktury.data_faktury,Faktury.kwota_calosci,Faktury.zaplacono,Platnosci.data_platnosci,Platnosci.kwota AS kwota_platnosci,Platnosci.metoda_platnosci,Samochody.marka,Samochody.model,Samochody.numer_rejestracyjny FROM Klienci JOIN Wypozyczenia ON Klienci.id_klienta = Wypozyczenia.id_klienta JOIN Faktury ON Wypozyczenia.id_wypozyczenia = Faktury.id_wypozyczenia LEFT JOIN Platnosci ON Faktury.id_faktury = Platnosci.id_faktury JOIN Samochody ON Wypozyczenia.id_samochodu = Samochody.id_samochodu WHERE Faktury.zaplacono = 'nie';
SELECT Samochody.marka,Samochody.model,Samochody.numer_rejestracyjny,Serwisy.opis,YEAR(Serwisy.data_serwisu) AS rok,Serwisy.koszt FROM Samochody JOIN Serwisy ON Samochody.id_samochodu = Serwisy.id_samochodu WHERE Serwisy.opis LIKE '%oleju%' OR Serwisy.opis LIKE '%filtrów%' ORDER BY Serwisy.koszt DESC;
SELECT    Klienci.imie,Klienci.nazwisko,COUNT(Wypozyczenia.id_wypozyczenia) AS liczba_wypozyczen,SUM(Wypozyczenia.kwota) AS laczna_kwota FROM Klienci JOIN Wypozyczenia ON Klienci.id_klienta = Wypozyczenia.id_klienta WHERE Wypozyczenia.data_wypozyczenia >= DATE_SUB(CURDATE(), INTERVAL 12 MONTH) AND Wypozyczenia.data_wypozyczenia <= CURDATE() GROUP BY Klienci.id_klienta, Klienci.imie, Klienci.nazwisko ORDER BY laczna_kwota DESC;
SELECT Klienci.id_klienta,Klienci.imie,Klienci.nazwisko,COUNT(Wypozyczenia.id_wypozyczenia) AS liczba_wypozyczen FROM Klienci JOIN Wypozyczenia ON Klienci.id_klienta = Wypozyczenia.id_klienta JOIN Samochody ON Wypozyczenia.id_samochodu = Samochody.id_samochodu WHERE Samochody.marka = 'BMW' AND Wypozyczenia.data_wypozyczenia >= DATE_SUB(CURDATE(), INTERVAL 2 YEAR) GROUP BY Klienci.id_klienta,Klienci.imie,Klienci.nazwisko HAVING COUNT(Wypozyczenia.id_wypozyczenia) > 3 ORDER BY Klienci.nazwisko ASC;
SELECT DISTINCT Samochody.id_samochodu,Samochody.marka,Samochody.model,Samochody.rok_produkcji FROM Samochody JOIN Wypozyczenia ON Samochody.id_samochodu = Wypozyczenia.id_samochodu ORDER BY Samochody.marka ASC;
SELECT Klienci.id_klienta,Klienci.imie,Klienci.nazwisko,Wypozyczenia.data_wypozyczenia,DATEDIFF(CURDATE(), Wypozyczenia.data_wypozyczenia) AS liczba_dni FROM Klienci JOIN Wypozyczenia ON Klienci.id_klienta = Wypozyczenia.id_klienta WHERE Wypozyczenia.status = 'wypożyczony' AND DATEDIFF(CURDATE(), Wypozyczenia.data_wypozyczenia) > 30 ORDER BY Klienci.nazwisko ASC;
