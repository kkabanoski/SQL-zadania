SELECT k.id_klienta,k.imie,k.nazwisko,ROUND(AVG(o.ocena), 2) AS srednia_wystawionych_ocen,COUNT(o.id_opinii) AS liczba_opinii FROM Klienci k JOIN Opinie o ON k.id_klienta = o.id_klienta GROUP BY k.id_klienta, k.imie, k.nazwisko HAVING AVG(o.ocena) >= 4 ORDER BY srednia_wystawionych_ocen DESC;

SELECT f.id_faktury,f.id_wypozyczenia,f.data_faktury,f.kwota_calosci AS kwota_faktury,f.zaplacono AS status_faktury,p.id_platnosci,p.data_platnosci,p.kwota AS kwota_wplaty,p.metoda_platnosci FROM Faktury f LEFT JOIN Platnosci p ON f.id_faktury = p.id_faktury ORDER BY f.id_faktury ASC;

SELECT s.id_samochodu, s.marka, s.model, COUNT(w.id_wypozyczenia) AS liczba_wypozyczen FROM Samochody s JOIN Wypozyczenia w ON s.id_samochodu = w.id_samochodu GROUP BY s.id_samochodu, s.marka, s.model ORDER BY liczba_wypozyczen DESC LIMIT 10;
SELECT k.id_klienta, k.imie, k.nazwisko, AVG(w.kwota) AS srednia_wartosc_wypozyczen, COUNT(w.id_wypozyczenia) AS liczba_wypozyczen FROM Klienci k JOIN Wypozyczenia w ON k.id_klienta = w.id_klienta GROUP BY k.id_klienta, k.imie, k.nazwisko HAVING COUNT(w.id_wypozyczenia) > 5 ORDER BY srednia_wartosc_wypozyczen DESC;
SELECT k.id_klienta,k.imie,k.nazwisko,COUNT(w.id_wypozyczenia) AS liczba_wypozyczen FROM Klienci k JOIN Wypozyczenia w ON k.id_klienta = w.id_klienta JOIN Samochody s ON w.id_samochodu = s.id_samochodu WHERE w.data_wypozyczenia >= DATE_SUB(CURDATE(),INTERVAL 1 YEAR) GROUP BY k.id_klienta,k.imie,k.nazwisko ORDER BY liczba_wypozyczen DESC;
SELECT s.marka,s.model,AVG(DATEDIFF(w.data_zwrotu,w.data_wypozyczenia)) AS srednia_liczba_dni,COALESCE((SELECT SUM(sr.koszt) FROM Serwisy sr WHERE sr.id_samochodu = s.id_samochodu),0) AS koszt_napraw FROM Samochody s JOIN Wypozyczenia w ON s.id_samochodu = w.id_samochodu GROUP BY s.id_samochodu,s.marka,s.model HAVING COUNT(w.id_wypozyczenia) > 5;

SELECT k.id_klienta, k.imie, k.nazwisko, COUNT(w.id_wypozyczenia) AS liczba_wypozyczen FROM Klienci k JOIN Wypozyczenia w ON k.id_klienta = w.id_klienta GROUP BY k.id_klienta, k.imie, k.nazwisko HAVING COUNT(w.id_wypozyczenia) >= 3 AND NOT EXISTS ( SELECT 1 FROM Opinie o WHERE o.id_klienta = k.id_klienta );
SELECT s.id_samochodu, s.marka, s.model, k.imie, k.nazwisko, DATEDIFF( COALESCE(w.data_zwrotu, CURDATE()), w.data_wypozyczenia ) AS liczba_dni FROM Samochody s JOIN Wypozyczenia w ON s.id_samochodu = w.id_samochodu JOIN Klienci k ON w.id_klienta = k.id_klienta WHERE w.data_wypozyczenia >= DATE_SUB(CURDATE(), INTERVAL 12 MONTH) AND w.data_wypozyczenia <= CURDATE() AND k.data_rejestracji >= '2022-01-01' AND k.data_rejestracji < '2024-01-01' ORDER BY liczba_dni DESC;
SELECT k.imie,k.nazwisko,k.email,f.id_faktury,f.data_faktury,f.kwota_calosci,f.zaplacono,p.data_platnosci,p.kwota AS kwota_platnosci,p.metoda_platnosci,s.marka,s.model,s.numer_rejestracyjny FROM Klienci k JOIN Wypozyczenia w ON k.id_klienta = w.id_klienta JOIN Faktury f ON w.id_wypozyczenia = f.id_wypozyczenia LEFT JOIN Platnosci p ON f.id_faktury = p.id_faktury JOIN Samochody s ON w.id_samochodu = s.id_samochodu WHERE f.zaplacono = 'nie';
SELECT s.marka,s.model,s.numer_rejestracyjny,sr.opis,YEAR(sr.data_serwisu) AS rok,sr.koszt FROM Samochody s JOIN Serwisy sr ON s.id_samochodu = sr.id_samochodu WHERE sr.opis LIKE '%oleju%' OR sr.opis LIKE '%filtrów%' ORDER BY sr.koszt DESC;
SELECT k.imie,k.nazwisko,COUNT(w.id_wypozyczenia) AS liczba_wypozyczen,SUM(w.kwota) AS laczna_kwota FROM Klienci k JOIN Wypozyczenia w ON k.id_klienta = w.id_klienta WHERE w.data_wypozyczenia >= DATE_SUB(CURDATE(),INTERVAL 12 MONTH) AND w.data_wypozyczenia <= CURDATE() GROUP BY k.id_klienta,k.imie,k.nazwisko ORDER BY laczna_kwota DESC;
SELECT k.id_klienta, k.imie, k.nazwisko, COUNT(w.id_wypozyczenia) AS liczba_wypozyczen FROM Klienci k JOIN Wypozyczenia w ON k.id_klienta = w.id_klienta JOIN Samochody s ON w.id_samochodu = s.id_samochodu WHERE s.marka = 'BMW' AND w.data_wypozyczenia >= DATE_SUB(CURDATE(), INTERVAL 2 YEAR) AND w.data_wypozyczenia <= CURDATE() GROUP BY k.id_klienta, k.imie, k.nazwisko HAVING COUNT(w.id_wypozyczenia) > 3 ORDER BY k.nazwisko ASC;
SELECT s.id_samochodu, s.marka, s.model, COUNT(DISTINCT w.id_klienta) AS liczba_klientow, COUNT(DISTINCT YEAR(w.data_wypozyczenia)) AS liczba_okresow FROM Samochody s JOIN Wypozyczenia w ON s.id_samochodu = w.id_samochodu GROUP BY s.id_samochodu, s.marka, s.model HAVING COUNT(DISTINCT YEAR(w.data_wypozyczenia)) > 1 ORDER BY s.marka ASC;
SELECT k.id_klienta,k.imie,k.nazwisko,w.data_wypozyczenia,DATEDIFF(CURDATE(),w.data_wypozyczenia) AS liczba_dni FROM Klienci k JOIN Wypozyczenia w ON k.id_klienta = w.id_klienta WHERE w.status = 'wypożyczony' AND DATEDIFF(CURDATE(),w.data_wypozyczenia) > 30 ORDER BY k.nazwisko ASC;
